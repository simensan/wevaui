# Consume the published archives without adding the plugin's CMake project:
# transitive dependencies in its build tree must not mask missing artifacts.
cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
project(weva_static_consumer CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_DIRECTORY "${WEVA_PLUGIN_DIR}")
    message(FATAL_ERROR "Set WEVA_PLUGIN_DIR to the static plugin artifact directory")
endif()
add_executable(weva_core_static_test ../load_test.cpp)
target_compile_definitions(weva_core_static_test PRIVATE WEVA_STATIC_PLUGIN)
target_include_directories(weva_core_static_test PRIVATE ../../src ../../../../libweva/include)
foreach(archive weva_core weva weva_icu_i18n weva_icu_common)
    # Restrict to static archives even if a previous shared build is alongside.
    if(MSVC)
        set(filename "${archive}.lib")
    elseif(archive STREQUAL "weva_core")
        set(filename "weva_core.a")
    else()
        set(filename "lib${archive}.a")
    endif()
    set(path "${WEVA_PLUGIN_DIR}/${filename}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing plugin archive: ${path}")
    endif()
    target_link_libraries(weva_core_static_test PRIVATE "${path}")
endforeach()
if(WIN32)
    target_link_libraries(weva_core_static_test PRIVATE advapi32)
elseif(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(weva_core_static_test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
